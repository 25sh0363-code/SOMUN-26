    24→async function sb(path, opts = {}) {
    25→  const res = await fetch(`${CONFIG.SUPABASE_URL}/rest/v1/${path}`, {
    26→    ...opts,
    27→    headers: {
    28→      apikey: CONFIG.SUPABASE_ANON_KEY,
    29→      Authorization: `Bearer ${CONFIG.SUPABASE_ANON_KEY}`,
    30→      "Content-Type": "application/json",
    31→      ...(opts.headers || {}),
    32→    },
    33→  });
    34→  if (!res.ok) {
    35→    const data = await res.json().catch(() => ({}));
    36→    throw new Error(data.message || `Supabase request failed (${res.status})`);
    37→  }
    38→  return res.json();
    39→}

   658→  return deck.slides[0] ? deck.slides[0].offsetWidth : window.innerWidth;

   661→function deckOnScroll() {
   662→  if (!deck.el) return;
   663→  const rect = deck.el.getBoundingClientRect();
   664→  const total = deck.el.offsetHeight - window.innerHeight;
   665→  if (total <= 0) return;
   666→  const p = Math.min(0.99999, Math.max(0, -rect.top / total));
   667→  const t = Math.round(p * (deck.n - 1));
   668→  if (t !== deck.active) {
   669→    deck.active = t;
   670→    deck.target = t * slideWidth();
   671→    deckSetCurrent(t);
   672→  }
   673→}

   675→/* spring integrator — stiffness 110, damping 24, mass 0.9 (as before) */
   676→function deckFrame(t) {
   677→  const last = deckFrame.last || t;
   678→  const dt = Math.min(0.05, (t - last) / 1000);
   679→  deckFrame.last = t;
   680→  const k = 110, c = 24, m = 0.9;
   681→  const F = -k * (deck.pos - deck.target) - c * deck.vel;
   682→  deck.vel += (F / m) * dt;
   683→  deck.pos += deck.vel * dt;
   684→  deck.track.style.transform = `translate3d(${-deck.pos}px,0,0)`;
   685→  requestAnimationFrame(deckFrame);
   686→}

   695→    deck.vel = 0;
